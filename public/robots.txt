# El muro son mensajes de personas, firmados y publicados sin revisión previa. El
# pie promete borrarlos a quien lo pida, y esa promesa no se puede cumplir dentro

Disallow: /api/messages

# Geocodificación: es un proxy con límite de uso, no una API para rastrear.
Disallow: /api/geocode

Disallow: /api/admin/
Disallow: /admin

# Los vuelcos de /d/ NO se bloquean aquí, y es a propósito: la hoja de cálculo
# los lee con IMPORTDATA, que respeta este fichero, así que un Disallow le
# impedía leer la URL que se le pide leer y devolvía #N/A. Quien los protege es
# la clave impredecible de la ruta; que no se indexen lo resuelve la cabecera
# X-Robots-Tag: noindex de la propia respuesta, que además funciona mejor —
# bloquear aquí el rastreo impediría al buscador llegar a leer ese noindex.
